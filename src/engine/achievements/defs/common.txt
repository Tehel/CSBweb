# CSBweb achievements 1
#
# The achievements every data set has (doc/ACHIEVEMENTS.md). A set's own achievements.txt
# adds to them, replaces one by giving its id, or switches one off with "disabled". An
# achievement that names a value, {name}, exists only in a set whose file defines it.
# The names are FTL's objects; one a set's objects do not have matches nothing there.

# Where the party is: the level it stands on at each tick's end. levels counts the maps of
# the dungeon that hold a floor, a corridor square; explored is the share of those squares
# the party has stood on, in percent (doc/ACHIEVEMENTS.md -> "The events").

[tourist]
name: Tourist
text: Set foot on every level of the dungeon.
tier: easy
when: levels > 0 and distinct(enter.level) >= levels

[cartographer]
name: Cartographer
text: Walk on 90% of the dungeon's floor.
tier: hard
when: explored >= 90

# The champions at each tick's end: party(where ...) counts those its filter accepts, by
# what each wears and holds, their sex, their rank in each class, 0 to 15 (8 EXPERT, 9 to
# 14 the six masters, 15 ARCHMASTER), and their load, in percent of the most they can
# carry (doc/ACHIEVEMENTS.md -> "The champions").

[low-profile]
name: Low Profile
text: Win without a champion ever rising above Expert in any class.
tier: hard
when: won and max(party(where fighter > 8 or ninja > 8 or priest > 8 or wizard > 8)) = 0

[archmaster]
name: Archmaster
text: Raise a champion to Archmaster in any class.
tier: hard
when: party(where fighter >= 15 or ninja >= 15 or priest >= 15 or wizard >= 15) >= 1

[jack-of-all-trades]

tier: hard

# cleanSweepLevels: the levels worth sweeping, a list; a monster that no blow kills, such
# as Lord Chaos, keeps his level from ever being swept.

[clean-sweep]
name: Clean Sweep
text: Leave a level with no monster alive on it.
tier: hard
when: ever(leave where cleared = yes and level in {cleanSweepLevels})
